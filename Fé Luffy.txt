local function cameraShake()
	local tempo = 0.4
	local intensidade = 0.5
	local startTime = tick()
	local connection
	connection = RunService.RenderStepped:Connect(function()
		if tick() - startTime > tempo then connection:Disconnect() return end
		local randomX = (math.random() - 0.1) * intensidade
		local randomY = (math.random() - 0.5) * intensidade
		local randomZ = (math.random() - 0.9) * intensidade
		camera.CFrame = camera.CFrame * CFrame.new(randomX, randomY, randomZ)
	end)
end

local function createBubble(hrp)
	local bubble = Instance.new("Part")
	bubble.Shape = Enum.PartType.Ball
	bubble.Material = Enum.Material.ForceField
	bubble.Color = Color3.fromRGB(255,0,0)
	bubble.Transparency = 0.5
	bubble.Anchored = true
	bubble.CanCollide = false
	bubble.Size = Vector3.new(2,2,2)
	bubble.CFrame = hrp.CFrame
	bubble.Parent = workspace
	TweenService:Create(bubble, TweenInfo.new(CONFIG_TROVAO.Duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Size = Vector3.new(250,250,250), Transparency = 1}):Play()
	Debris:AddItem(bubble, CONFIG_TROVAO.Duration + 0.2)
end

local function createRealisticRay(startPos, direction)
	local dir = direction or Vector3.new(math.random(-150,150), math.random(-100,100), math.random(-150,150)).Unit
	local length = math.random(80,200)
	local segments = math.random(8,14)
	local points = {startPos}
	for i = 1, segments do
		local t = i/segments
		local offset = Vector3.new(math.sin(t*math.pi*4)*math.random(2,6), math.cos(t*math.pi*3)*math.random(2,6), math.sin(t*math.pi*5)*math.random(2,6))
		local nextPos = startPos + dir*(length*t) + offset*(1-t)*8
		table.insert(points, nextPos)
	end
	for i = 1, #points-1 do
		local a = points[i]
		local b = points[i+1]
		local dist = (b-a).Magnitude
		if dist > 0.2 then
			local part = Instance.new("Part")
			part.Anchored = true
			part.CanCollide = false
			part.Material = Enum.Material.Neon
			part.Color = Color3.fromRGB(255,0,0)
			part.Size = Vector3.new(3,3,dist)
			part.CFrame = CFrame.new(a,b) * CFrame.new(0,0,-dist/2)
			part.Parent = workspace
			local light = Instance.new("PointLight")
			light.Color = Color3.fromRGB(255,50,50)
			light.Range = 12
			light.Brightness = 1.5
			light.Parent = part
			TweenService:Create(part, TweenInfo.new(CONFIG_TROVAO.RayLifespan-0.5), {Transparency = 1}):Play()
			Debris:AddItem(part, CONFIG_TROVAO.RayLifespan)
			Debris:AddItem(light, CONFIG_TROVAO.RayLifespan)
		end
	end
end

local function createSkyRay(hrp)
	local startPos = hrp.Position + Vector3.new(math.random(-300,300), math.random(120,250), math.random(-300,300))
	local direction = (hrp.Position - startPos).Unit
	createRealisticRay(startPos, direction)
end

local function createPlayerRays(hrp)
	for i = 1, 3 do
		local dir = Vector3.new(math.random(-200,200), math.random(-80,150), math.random(-200,200)).Unit
		createRealisticRay(hrp.Position, dir)
	end
end

local activeTrovao = false
local function startRaysLoop(hrp)
	while activeTrovao and hrp and hrp.Parent do
		createPlayerRays(hrp)
		if math.random() < 0.5 then createSkyRay(hrp) end
		task.wait(CONFIG_TROVAO.RayRate)
	end
end

local function activateTrovao(hrp)
	if activeTrovao or not hrp then return end
	activeTrovao = true
	createBubble(hrp)
	cameraShake()
	ExecuteSoundGlobal(CONFIG_TROVAO.SoundID)
	task.spawn(function() startRaysLoop(hrp) end)
	task.wait(CONFIG_TROVAO.Duration)
	activeTrovao = false
end

botaoTrovao.MouseButton1Click:Connect(function()

	local hrp = character:FindFirstChild("HumanoidRootPart")
	if not hrp then return end
	activateTrovao(hrp)
	onCooldownTrovao = true
	overlayTrovao.Visible = true
	local startTime = tick()
	local conn

		if remaining <= 0 then
			overlayTrovao.Visible = false
			onCooldownTrovao = false
			conn:Disconnect()
		end
	end)
end)

		TweenService:Create(Botao, TweenInfo.new(0.1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Size = UDim2.new(0, 48, 0, 48)}):Play()
		task.wait(0.1)
		TweenService:Create(Botao, tweenInfo, {Size = UDim2.new(0, 44, 0, 44)}):Play()
	end)

	local function setAtivo(ativo)
		if ativo then
			Botao.BackgroundColor3 = corAtivo or Color3.fromRGB(80, 200, 80)
			Botao.BackgroundTransparency = 0.1
		else
			Botao.BackgroundColor3 = Color3.fromRGB(40, 40, 45)
			Botao.BackgroundTransparency = 0.15
		end
	end

	return Botao, setAtivo
end

-- Módulo Elástico
local elasticActiveFlag = false
local elasticTiles = {}
local bounceForceMin = 100
local bounceForceMax = 350
local checkRate = 0.03
local sinkAmount = 2
local tweenTime = 0.25

local function makeElastic(part, hrp)
	if elasticTiles[part] then return end
	elasticTiles[part] = {originalCFrame = part.CFrame}
	local conn = part.Touched:Connect(function(hit)
		if hit.Parent == character and hrp and hrp.Parent then
			local bounceForce = math.random(bounceForceMin, bounceForceMax)
			local bv = Instance.new("BodyVelocity")
			bv.Velocity = Vector3.new(0, bounceForce, 0)
			bv.MaxForce = Vector3.new(0, math.huge, 0)
			bv.P = 1500
			bv.Parent = hrp
			Debris:AddItem(bv, 0.25)
			local downCFrame = part.CFrame * CFrame.new(0, -sinkAmount, 0)
			local tweenDown = TweenService:Create(part, TweenInfo.new(tweenTime, Enum.EasingStyle.Elastic, Enum.EasingDirection.Out), {CFrame = downCFrame})
			local tweenUp = TweenService:Create(part, TweenInfo.new(tweenTime*1.2, Enum.EasingStyle.Bounce, Enum.EasingDirection.Out), {CFrame = elasticTiles[part].originalCFrame})
			tweenDown:Play()
			tweenDown.Completed:Wait()
			tweenUp:Play()
		end
	end)
	elasticTiles[part].connection = conn
end

local function removeElastic()
	for part, data in pairs(elasticTiles) do
		if data.connection then data.connection:Disconnect() end
		if part and part.Parent then
			part.CFrame = data.originalCFrame
		end
	end
	elasticTiles = {}
end

local function checkFloorLoop(hrp)
	while elasticActiveFlag and hrp and hrp.Parent do
		local raycastParams = RaycastParams.new()
		raycastParams.FilterDescendantsInstances = {character}
		raycastParams.FilterType = Enum.RaycastFilterType.Blacklist
		raycastParams.IgnoreWater = true
		local rayResult = workspace:Raycast(hrp.Position, Vector3.new(0, -5, 0), raycastParams)
		if rayResult and rayResult.Instance.Anchored then
			makeElastic(rayResult.Instance, hrp)
		end
		task.wait(checkRate)
	end
end

local function startElastic()
	if elasticActiveFlag then return end
	elasticActiveFlag = true
	humanoid.WalkSpeed = 30
	local hrp = character:FindFirstChild("HumanoidRootPart")
	if hrp then
		task.spawn(function() checkFloorLoop(hrp) end)
	end
end

local function stopElastic()
	if not elasticActiveFlag then return end
	elasticActiveFlag = false
	humanoid.WalkSpeed = 16
	removeElastic()
end

-- Módulo Fumaça (com quantidades REDUZIDAS)
local smokeActiveFlag = false
local smokeEmitters = {}
local smokeRenderConnection = nil

-- Template de partículas ajustado para menos fumaça
local vaporToggleTemplate = Instance.new("ParticleEmitter")
vaporToggleTemplate.Texture = "rbxassetid://258128463"
vaporToggleTemplate.Speed = NumberRange.new(1, 2)           -- antes 2~3
vaporToggleTemplate.Lifetime = NumberRange.new(0.4, 0.6)    -- antes 0.8~1.2
vaporToggleTemplate.Rate = 14                               -- antes 40
vaporToggleTemplate.Size = NumberSequence.new({NumberSequenceKeypoint.new(0, 0.8), NumberSequenceKeypoint.new(1, 1.0)})  -- tamanho reduzido
vaporToggleTemplate.Transparency = NumberSequence.new({NumberSequenceKeypoint.new(0, 0.4), NumberSequenceKeypoint.new(1, 1)})  -- mais transparente no início
vaporToggleTemplate.Color = ColorSequence.new(Color3.new(1,1,1))
vaporToggleTemplate.EmissionDirection = Enum.NormalId.Top

local function createSmokeEmitters(char)
	local bodyParts = {}
	for _, part in pairs(char:GetDescendants()) do
		if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
			table.insert(bodyParts, part)
		end
	end
	smokeEmitters = {}
	for _, part in ipairs(bodyParts) do
		local puffPart = Instance.new("Part")
		puffPart.Size = Vector3.new(1,1,1)
		puffPart.Anchored = true
		puffPart.CanCollide = false
		puffPart.Transparency = 1
		puffPart.CFrame = part.CFrame
		puffPart.Parent = workspace
		local puff = vaporToggleTemplate:Clone()
		puff.Parent = puffPart
		puff.Enabled = smokeActiveFlag
		table.insert(smokeEmitters, {part=part, puffPart=puffPart, emitter=puff})
	end
end

local function destroySmokeEmitters()
	for _, e in ipairs(smokeEmitters) do
		if e.emitter then e.emitter:Destroy() end
		if e.puffPart then Debris:AddItem(e.puffPart, 0) end
	end
	smokeEmitters = {}
end

local function startSmoke()
	if smokeActiveFlag then return end
	smokeActiveFlag = true
	if player.Character then createSmokeEmitters(player.Character) end
	if not smokeRenderConnection then
		smokeRenderConnection = RunService.RenderStepped:Connect(function()
			if smokeActiveFlag then
				for _, e in ipairs(smokeEmitters) do
					if e.part and e.puffPart and e.part.Parent then
						e.puffPart.CFrame = e.part.CFrame
					end
				end
			end
		end)
	end
end

local function stopSmoke()
	if not smokeActiveFlag then return end
	smokeActiveFlag = false
	destroySmokeEmitters()
	if smokeRenderConnection then
		smokeRenderConnection:Disconnect()
		smokeRenderConnection = nil
	end
end
